# ===========================================================================
#  Veri-Fact Desktop - Python dependencies
# ---------------------------------------------------------------------------
#  Install everything:
#      pip install -r requirements.txt
#
#  Then download the spaCy language model (a separate step - it is NOT a pip
#  package listed below):
#      python -m spacy download en_core_web_sm
#
#  Python 3.10 or newer is required.
#  tkinter is NOT listed here: it ships with the official python.org installer.
#  On Debian/Ubuntu install it with `sudo apt install python3-tk`.
# ===========================================================================


# ---------------------------------------------------------------------------
#  CORE - required for the main features
# ---------------------------------------------------------------------------

# NLP extraction (named-entity recognition for dates, places and quantities).
# The app degrades gracefully to a built-in regex extractor if this or its
# model is unavailable - check Help > Backend Diagnostics to see which is live.
spacy>=3.7,<4.0

# REQUIRED BY SPACY, BUT NOT DECLARED BY IT.
# ------------------------------------------------------------------
# spaCy 3.8.x imports `click` directly (spacy/cli/_util.py) but only depends on
# it transitively through `typer`. typer 0.27.0 REMOVED its click dependency,
# so a fresh `pip install spacy` now succeeds while `import spacy` and
# `python -m spacy download en_core_web_sm` both fail with:
#     ModuleNotFoundError: No module named 'click'
# Pinning click here makes the documented install work on a clean machine.
# Remove once spaCy declares click itself (tracked upstream: explosion/spaCy).
click>=8.0,<9.0

# CSV and Excel parsing.
pandas>=2.0,<3.0

# Reading .xlsx via pandas AND writing the colour-coded Excel reports.
openpyxl>=3.1,<4.0


# ---------------------------------------------------------------------------
#  OPTIONAL - faster fuzzy matching for smart-suggest
# ---------------------------------------------------------------------------
# Smart-suggest ("Did you mean 'active_users'?") works out of the box using
# difflib from the Python standard library. Installing this C extension makes
# matching roughly 10x faster on large metric vocabularies. Results are
# identical either way, so this is purely a performance option.
#
# To skip it, delete or comment out the line below.
python-Levenshtein>=0.25 ; platform_python_implementation == "CPython"


# ---------------------------------------------------------------------------
#  OPTIONAL - external enterprise databases
# ---------------------------------------------------------------------------
# All three imports are wrapped in try/except, so the application starts
# perfectly well without them and simply uses the bundled SQLite database.
# Uncomment only the driver you actually need.

# PostgreSQL
# psycopg2-binary>=2.9,<3.0

# MySQL / MariaDB
# pymysql>=1.1,<2.0

# Microsoft SQL Server (also needs the Microsoft ODBC Driver installed on the OS)
# pyodbc>=5.0,<6.0


# ---------------------------------------------------------------------------
#  OPTIONAL - drag-and-drop files onto the window
# ---------------------------------------------------------------------------
# Without this the Import Files button still works normally.
# tkinterdnd2>=0.4,<0.5


# ---------------------------------------------------------------------------
#  DELIBERATELY NOT REQUIRED
# ---------------------------------------------------------------------------

# Indian 12,34,567.89 / spaced 1 234 567,89) are handled by verifact/numbers.py
# using the standard library only.
#
# Derived-metric formulas are evaluated by verifact/formulas.py using Python's
